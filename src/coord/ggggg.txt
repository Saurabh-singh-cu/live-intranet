import React, { useEffect, useState } from "react";
import styles from "./ProposedCalendarCoord.module.css";
import apiClient from "../config/apiClient";

const card = ({ item, isActive, onClick}) => {
  return (
    <div></div>
  )
}

const ProposedCalenderCoord = () => {
  const [entityData, setEntityData] = useState([]);

  const getUserId = localStorage.getItem("user")
    ? JSON.parse(localStorage.getItem("user")).department_id
    : "";

  console.log(getUserId);
  const apiCall = async (getUserId) => {
    const response = await apiClient.get(
      `entity-proposed-details/${getUserId}/`
    );
    setEntityData(response?.data?.registration_details);

    console.log(response?.data?.registration_details, "RRRRR");
  };

  useEffect(() => {
    apiCall(getUserId);
  }, [getUserId]);

  return (
    <>
      <div className={styles.container}>
        <div className={styles.leftColumn}>
          <h2>Entity Registration</h2>

          <div>
            {entityData &&
              entityData?.map((item, index) => (
                <div
                  className={`${styles.card} ${isActive ? styles.active : ""}`}
                  // onClick={onClick}
                >
                  <h3 className={styles.cardTitle}>
                    {item.registeration_name}
                  </h3>
                </div>
              ))}
          </div>
        </div>
      </div>
    </>
  );
};

export default ProposedCalenderCoord;
